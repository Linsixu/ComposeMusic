-- 1. 机构表 (Institutions) - [修改] 新增联系电话
CREATE TABLE institutions (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(100) NOT NULL,
    contact_phone VARCHAR(20), -- 新增字段：联系电话
    address VARCHAR(255),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
— 1.1 名字添加唯一键
ALTER TABLE institutions ADD CONSTRAINT uk_institutions_name UNIQUE (name);

-- 2. 老师表 (Teachers)
CREATE TABLE teachers (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    institution_id BIGINT NOT NULL,
    name VARCHAR(50) NOT NULL,
    subject VARCHAR(50),
    phone VARCHAR(20),
    FOREIGN KEY (institution_id) REFERENCES institutions(id),
    INDEX idx_teachers_inst (institution_id)
);

-- 3. 学生表 (Students)
CREATE TABLE students (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    institution_id BIGINT NOT NULL,
    name VARCHAR(50) NOT NULL,
    phone VARCHAR(20),
    FOREIGN KEY (institution_id) REFERENCES institutions(id),
    INDEX idx_students_inst (institution_id),
    UNIQUE KEY uk_students_name_phone_inst (name, phone, institution_id)
);

-- 4. 课程模板表 (Course Templates)
CREATE TABLE course_templates (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    teacher_id BIGINT NOT NULL,
    institution_id BIGINT NOT NULL,
    title VARCHAR(100) NOT NULL,
    duration_minutes INT DEFAULT 60,
     subject  VARCHAR(50) DEFAULT NULL COMMENT '授课科目’,
    description TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (teacher_id) REFERENCES teachers(id),
    FOREIGN KEY (institution_id) REFERENCES institutions(id),
    INDEX idx_templates_inst (institution_id, teacher_id)
);

ALTER TABLE course_templates ADD COLUMN subject VARCHAR(50) DEFAULT NULL COMMENT '授课科目';

-- 5. 具体课时表 (Class Sessions)
CREATE TABLE class_sessions (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    template_id BIGINT NOT NULL,
    teacher_id BIGINT NOT NULL,
    institution_id BIGINT NOT NULL,
    start_time_ms BIGINT NOT NULL,
    duration INT NOT NULL,
    max_capacity INT DEFAULT 1,
    current_booked INT DEFAULT 0,
    status TINYINT NOT NULL DEFAULT 0,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (template_id) REFERENCES course_templates(id),
    FOREIGN KEY (teacher_id) REFERENCES teachers(id),
    INDEX idx_sessions_query (institution_id, status, start_time_ms)
);


-- 新增唯一约束（命名为 uk_institution_status_time）
ALTER TABLE class_sessions
ADD CONSTRAINT uk_institution_start_time
UNIQUE KEY (institution_id, teacher_id, start_time_ms);

-- 6. 预约记录表 (Reservations)
CREATE TABLE reservations (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    session_id BIGINT NOT NULL,
    student_id BIGINT NOT NULL,
    institution_id BIGINT NOT NULL,
    teacher_id BIGINT NOT NULL,
    status TINYINT NOT NULL DEFAULT 1,
    booked_at_ms BIGINT NOT NULL,
    FOREIGN KEY (session_id) REFERENCES class_sessions(id),
    FOREIGN KEY (student_id) REFERENCES students(id),
    UNIQUE KEY uk_session_student (session_id, student_id),
    INDEX idx_reservations_student (student_id, status),
    INDEX idx_reservations_session (session_id)
);
